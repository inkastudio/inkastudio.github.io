Sub-Millisecond Input & Velocity Verification

Before any 3D character mesh is imported, our movement controller is stress-tested across 10,000 simulated collision frames. 

[BurstCompile(FloatPrecision.Standard)]
public struct SplatSurfaceJob : IJobParallelFor {
    // Deterministic zero-allocation surface sampling
    public void Execute(int index) {
        float inkDensity = SurfaceGrid[index];
        VelocityMultiplier[index] = math.lerp(0.65f, 1.35f, inkDensity);
    }
}
